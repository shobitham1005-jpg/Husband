        (function() {

            // get elements

            const noBtn = document.getElementById('noBtn');

            const yesBtn = document.getElementById('yesBtn');

            const messageDiv = document.getElementById('loveMessage');

            const panel = document.getElementById('buttonPanel');


            let moveAttempts = 0;  // just for fun, no logic


            // ----- make NO button move away on touch/click (for mobile & desktop) -----

            function moveNoButton(e) {

                // Prevent default to avoid any double tap zoom or accidental click

                e.preventDefault();

                e.stopPropagation();  // avoid triggering click if any


                // get panel dimensions and button dimensions

                const panelRect = panel.getBoundingClientRect();

                const btnRect = noBtn.getBoundingClientRect();


                // button width including margin? we use offsetWidth

                const btnWidth = noBtn.offsetWidth;

                const btnHeight = noBtn.offsetHeight;


                // max left offset within panel (with some margin)

                const maxX = panelRect.width - btnWidth - 8;  // 8px safe margin

                const maxY = panelRect.height - btnHeight - 8;


                // Ensure we don't go outside the panel (button must remain inside button-panel)

                // we use relative positioning inside panel (which is position relative)

                // but the button is already relatively positioned? we add position relative to panel

                // and make button position absolute? Let's do quick adjustment.

                // For easier moving, we set panel to relative and button to absolute.

                // Buttons are side by side – we'll adjust noBtn only.

                // However yesBtn stays in flow, so we need to wrap or change.

                // Quick fix: make panel relative, and make both buttons absolute? That would overlap.

                // Better: Keep buttons inline, but we can move No using transform (translate) inside flow?

                // Transform won't affect layout flow, so it can overlay yes or go out.

                // Let's adjust with position: relative on noBtn and use left, top.

                // Because panel is flex, using relative + left/top will shift from its original position but without breaking layout too much.

                // We'll also set a higher z-index.


                // generate random offsets within panel limits, but we want to escape the finger.

                // We calculate shift relative to its original position.

                // original position is (0,0) if we reset left/ top. We need to ensure it moves far enough.

                // For simplicity, we'll compute new position inside panel using absolute coordinates?

                // But easier: move with absolute inside panel (which we set to relative).

                // We'll change button to position: absolute so we can freely move.

                // But we need to keep yesBtn static and noBtn absolute inside same panel.

                // Set panel to relative, yesBtn stays static (but we need to give it some left margin).

                // This is getting complex. Let's do quick method: keep both inline, but on noBtn hover/touch we use transform: translate(x,y)

                // and the button moves visually, but click area stays? Actually transform moves the hit area as well.

                // Perfect. We'll use transform translate. We also ensure it moves far enough away from touch point.


                // Get touch/click coordinates relative to panel to push opposite direction

                let clientX, clientY;

                if (e.touches) {

                    clientX = e.touches[0].clientX;

                    clientY = e.touches[0].clientY;

                } else {

                    clientX = e.clientX;

                    clientY = e.clientY;

                }


                // get button center

                const btnCenterX = btnRect.left + btnWidth/2;

                const btnCenterY = btnRect.top + btnHeight/2;


                // vector from touch to button center

                let deltaX = btnCenterX - clientX;

                let deltaY = btnCenterY - clientY;


                // Normalize or just push strongly in same direction (if touch is near button)

                const distance = Math.sqrt(deltaX*deltaX + deltaY*deltaY);

                const strength = 200; // pixels to move away


                if (distance < 150) { // only if they try to tap nearby

                    // move away: opposite direction from touch point relative to button

                    // if touch is right of button, move button left (negative)

                    let moveX = 0, moveY = 0;


                    // decide direction based on where the touch happened

                    if (Math.abs(deltaX) > 20) {

                        moveX = deltaX > 0 ? strength : -strength; // if touch left, move right

                    } else {

                        moveX = (Math.random() - 0.5) * strength; // random if uncertain

                    }


                    if (Math.abs(deltaY) > 20) {

                        moveY = deltaY > 0 ? strength : -strength;

                    } else {

                        moveY = (Math.random() - 0.5) * strength;

                    }


                    // clamp movement so button stays roughly inside view (soft limit)

                    const currentTransform = window.getComputedStyle(noBtn).transform;

                    let currentX = 0, currentY = 0;

                    if (currentTransform !== 'none') {

                        const values = currentTransform.split('(')[1].split(')')[0].split(',');

                        if (values.length >= 6) {

                            currentX = parseFloat(values[4]);

                            currentY = parseFloat(values[5]);

                        }

                    }


                    let newX = currentX + moveX;

                    let newY = currentY + moveY;


                    // keep within 200px range from origin to avoid out of screen

                    newX = Math.min(180, Math.max(-180, newX));

                    newY = Math.min(140, Math.max(-140, newY));


                    noBtn.style.transform = `translate(${newX}px, ${newY}px)`;

                    noBtn.style.transition = 'transform 0.15s ease';

                    moveAttempts++;

                } else {

                    // if far, maybe reset to origin slowly? not needed

                }


                // Always cancel default to avoid zoom / scroll

                return false;

            }


            // attach both touch and mouse events

            noBtn.addEventListener('touchstart', moveNoButton, { passive: false });

            noBtn.addEventListener('mousedown', moveNoButton);

            // for mouse leave reset? but keep as is


            // also for 'click' on noBtn we want to prevent any default and still move

            noBtn.addEventListener('click', (e) => {

                e.preventDefault();

                e.stopPropagation();

                // move again to be safe

                moveNoButton(e);

                return false;

            });


            // ----- YES button behaviour -----

            yesBtn.addEventListener('click', (e) => {

                e.preventDefault();


                // show message

                messageDiv.classList.add('show');


                // optionally hide no button? but we can just let it be.

                // But we want to make noBtn less prominent? we can fade it.

                noBtn.style.opacity = '0.3';

                noBtn.style.pointerEvents = 'none';  // disable no after yes

                yesBtn.style.transform = 'scale(1.1)';

                yesBtn.style.backgroundColor = '#ffd9e6';


                // also maybe stop moving

                // add some extra hearts? already floating.

                // and disable further yes clicks? not necessary

                yesBtn.disabled = true;

                yesBtn.style.opacity = '0.9';

            });


            // reset no button's transform on window resize to avoid weird positions

            window.addEventListener('resize', () => {

                noBtn.style.transform = 'translate(0, 0)';

            });


            // optional: touchstart on yes to give haptic (but not needed)

            // also disable context menu on buttons

            noBtn.addEventListener('contextmenu', (e) => e.preventDefault());

            yesBtn.addEventListener('contextmenu', (e) => e.preventDefault());


            // Preload message hidden initially (default ok)

            // Also make sure noBtn can't be easily tapped: move also on touch move

            noBtn.addEventListener('touchmove', (e) => {

                e.preventDefault();

                moveNoButton(e);

            }, { passive: false });


            // For mobile, we also want to move if they drag toward button

            document.addEventListener('touchmove', (e) => {

                // optional: fine

            }, { passive: false });


            // one extra: reset no double tap zoom

        })();
